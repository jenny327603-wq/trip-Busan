<script>
// --- 請務必填入你的 Firebase 配置 ---
const firebaseConfig = {
  apiKey: "你的KEY",
  authDomain: "你的網域",
  projectId: "你的ID"
};

firebase.initializeApp(firebaseConfig);
const db = firebase.firestore();
const tripId = "korea_trip_001";

// 初始化圖表
let myChart;
function updateChart(chartData) {
    const ctx = document.getElementById('myChart').getContext('2d');
    if (myChart) myChart.destroy();
    myChart = new Chart(ctx, {
        type: 'doughnut',
        data: {
            labels: Object.keys(chartData),
            datasets: [{
                data: Object.values(chartData),
                backgroundColor: ['#FF6384', '#36A2EB', '#FFCE56', '#4BC0C0'],
                borderWidth: 0
            }]
        },
        options: { plugins: { legend: { display: false } } }
    });
}

// 新增資料
function add() {
    const item = document.getElementById('itemEl').value;
    const amount = Number(document.getElementById('amountEl').value);
    const rate = Number(document.getElementById('rateEl').value);
    const payer = document.getElementById('payerEl').value;
    const type = document.getElementById('typeEl').value;

    if (!item || !amount) return alert("請輸入完整資料");

    db.collection("trips").doc(tripId).collection("expenses").add({
        item, amount, twd: Math.round(amount * rate), payer, type, time: Date.now()
    }).then(() => {
        document.getElementById('itemEl').value = "";
        document.getElementById('amountEl').value = "";
    });
}

// 刪除資料
function deleteData(docId) {
    if(confirm("確定要刪除這筆紀錄嗎？")) {
        db.collection("trips").doc(tripId).collection("expenses").doc(docId).delete();
    }
}

// 實時監聽
db.collection("trips").doc(tripId).collection("expenses").orderBy("time", "desc").onSnapshot(snapshot => {
    let total = 0;
    let html = "";
    let chartStats = {};

    snapshot.forEach(doc => {
        const d = doc.data();
        const id = doc.id;
        total += d.twd;
        
        // 統計分類
        chartStats[d.type] = (chartStats[d.type] || 0) + d.twd;

        html += `
            <div class="list-group-item">
                <div>
                    <div class="fw-bold">${d.item} <span class="category-tag">${d.type}</span></div>
                    <small class="text-muted">₩${d.amount.toLocaleString()} (by ${d.payer})</small>
                </div>
                <div class="text-end">
                    <div class="fw-bold text-primary">NT$ ${d.twd.toLocaleString()}</div>
                    <button onclick="deleteData('${id}')" class="delete-btn">×</button>
                </div>
            </div>
        `;
    });

    document.getElementById('list').innerHTML = html;
    document.getElementById('totalTwd').innerText = `NT$ ${total.toLocaleString()}`;
    updateChart(chartStats);
});
